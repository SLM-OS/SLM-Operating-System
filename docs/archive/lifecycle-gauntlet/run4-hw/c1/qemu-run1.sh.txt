#!/bin/bash
# One QEMU ARM64 test-kernel run (lifecycle gauntlet run 4, C1, #1213).
# usage: run1.sh <outfile> <elf> <single|multi> <what> <head>
# QEMU command = `make -n test` (ARM64, QEMU_VIRT) with
# -accel tcg,tb-size=128,thread=<mode> added (as run 3's qemu-1213 runner),
# under systemd-run MemoryMax=3G CPUQuota=200% timeout 120, fresh 4G sparse
# SD image per run.
set -u
out=$1; elf=$2; mode=$3; what=$4; head=$5
tmp=$(mktemp -d /tmp/claude-1000/c1run.XXXXXX)
truncate -s 4G "$tmp/sd.img"
load0=$(cut -d' ' -f1-3 /proc/loadavg)
nq0=$(pgrep -c -f qemu-system || true)
start=$(date -u +%FT%TZ)
cmd=(systemd-run --user --scope -q -p MemoryMax=3G -p CPUQuota=200% timeout 120
     qemu-system-aarch64 -machine virt -cpu cortex-a76 -smp cores=4 -m 1G
     -accel tcg,tb-size=128,thread=$mode -nographic
     -global virtio-mmio.force-legacy=false -device virtio-net-device,netdev=net0 -netdev user,id=net0
     -device virtio-rng-pci,bus=pcie.0 -drive id=slmos-sd,if=none,format=raw,file=$tmp/sd.img
     -device sdhci-pci -device sd-card,drive=slmos-sd
     -semihosting -kernel "$elf")
"${cmd[@]}" > "$tmp/serial.log" 2>&1
rc=$?
end=$(date -u +%FT%TZ)
load1=$(cut -d' ' -f1-3 /proc/loadavg)
ser=$tmp/serial.log
banner=$(grep -a -m1 -o 'SLM-OS v[^)]*)' "$ser" || echo "(none)")
samples=$(grep -a -o 'Samples collected: isolated=[0-9]*/[0-9]*, normal=[0-9]*/[0-9]*' "$ser" | head -1)
nto=$(grep -a -c 'iso_latency_[a-z]*: TIMEOUT' "$ser")
ntrace=$(grep -a -c 'SLOTTRACE-DUMP-BEGIN' "$ser")
npanic=$(grep -a -c 'KERNEL PANIC' "$ser")
nfault=$(grep -a -c 'SLOTTRACE-FAULT\|PAGE FAULT' "$ser")
nfail=$(grep -a -c '\[FAIL\]' "$ser")
stress=$(grep -a -m1 -o 'slot_reuse_stress: [^\r]*' "$ser" || true)
if [ "$rc" = 0 ]; then res=PASS; elif [ "$rc" = 124 ]; then res=TIMEOUT; else res="FAIL(rc=$rc)"; fi
[ "$ntrace" -gt 0 ] && res="$res+TRACE-DUMP"
[ "$npanic" -gt 0 ] && res="$res+PANIC"
[ "$nto" -gt 0 ] && res="$res+ISO-TIMEOUTx$nto"
{
  echo "# $what; head $head; elf sha256 $(sha256sum "$elf" | cut -d' ' -f1); accel tcg thread=$mode; RESULT $res; qemu rc $rc; ${samples:-Samples collected: (not printed)}; [FAIL] lines $nfail; ${stress:-}"
  echo "# boot banner: $banner"
  echo "# started $start, finished $end; loadavg at start $load0, at end $load1; qemu-system processes on host at start $nq0"
  echo "# command: ${cmd[*]}"
  echo "# ---- complete QEMU output (serial, stdout+stderr) ----"
  cat "$ser"
} > "$out"
rm -rf "$tmp"
echo "$(basename "$out") $res ${samples:-}"
