#!/usr/bin/env python3
"""Per-run table for run 4 C1 QEMU files. usage: table.py <evidence dir>"""
import glob, os, re, sys
d = sys.argv[1]
groups = [
    ("qemu-1213-main-multi-run", "main + 1 s waits (01931af6), thread=multi"),
    ("qemu-1213-main-single-run", "main + 1 s waits (01931af6), thread=single"),
    ("qemu-1213-trace-multi-run", "01931af6 + SLOT_TRACE=ON (b8df6f5f), thread=multi"),
    ("qemu-1213-stress-unfixed-multi-run", "UNFIXED + stress test (884aab2d), thread=multi"),
    ("qemu-1213-stresstrace-multi-run", "UNFIXED + stress test, SLOT_TRACE=ON (884aab2d), thread=multi"),
    ("qemu-1213-fix-multi-run", "FIX (912efbd0): stress + 1 s waits, thread=multi"),
    ("qemu-1213-step5-multi-run", "FIX + permanent 1 s waits asserted (6aee62c1), thread=multi"),
    ("qemu-1211-unfixed-multi-run", "#1211 window, UNFIXED main + SLOT_TRACE=ON (8b0065ce), thread=multi"),
    ("qemu-1211-fix-multi-run", "#1211 window, FIX (2f043ffa), thread=multi"),
    ("qemu-1229-wstrace-v1-multi-run", "main WORK_STEALING=ON + SLOT_TRACE (6f405a76, first dump code), thread=multi"),
    ("qemu-1229-wstrace-multi-run", "main WORK_STEALING=ON + SLOT_TRACE (7e07b384), thread=multi"),
    ("qemu-1229-fix3wson-multi-run", "FIX d8b21ec3 (#1213+#1229, first #1229 test), WORK_STEALING=ON, thread=multi"),
    ("qemu-1229-fix3wsoff-multi-run", "FIX d8b21ec3, WORK_STEALING=OFF, thread=multi"),
    ("qemu-1229-pdunfixed-8f4aaffd-multi-run", "8f4aaffd without the #1229 cpu re-read (local), WS=OFF"),
    ("qemu-1229-pdunfixedws-8f4aaffd-multi-run", "8f4aaffd without the #1229 cpu re-read (local), WS=ON"),
    ("qemu-1229-pdunfixed-multi-run", "6d4fe041 without the #1229 cpu re-read (local), WS=OFF"),
    ("qemu-1229-pdunfixedws-multi-run", "6d4fe041 without the #1229 cpu re-read (local), WS=ON"),
    ("qemu-1229-fix5wson-multi-run", "FIX 6d4fe041, WORK_STEALING=ON, thread=multi"),
    ("qemu-1229-fix5wsoff-multi-run", "FIX 6d4fe041 , WORK_STEALING=OFF, thread=multi"),
    ("qemu-1229-fix6wson-multi-run", "FIX 9185d4cb (final head; test-only after 364544c1), WORK_STEALING=ON, thread=multi"),
    ("qemu-1229-fix6wsoff-multi-run", "FIX 9185d4cb (final head), WORK_STEALING=OFF, thread=multi"),
    ("qemu-5e5548a1-head-wsoff-multi-run", "FIX 5e5548a1 (switch-in wait reverted), default config WS=OFF"),
    ("qemu-5e5548a1-head-wson-multi-run", "FIX 5e5548a1, WORK_STEALING=ON (runs 01-10 alternated with main)"),
    ("qemu-1230-litpair-main-wson-multi-run", "main 6c1f4c9c WORK_STEALING=ON (alternated with 5e5548a1 runs 01-10)"),
    ("qemu-1230-lit-9185d4cb-wson-multi-run", "#1230 paired load check: 9185d4cb WS=ON (run alongside 6d4fe041)"),
    ("qemu-1230-lit-6d4fe041-wson-multi-run", "#1230 paired load check: 6d4fe041 WS=ON (run alongside 9185d4cb)"),
]
def g(pat, s, default='-'):
    m = re.search(pat, s)
    return m.group(1) if m else default
out = []
out.append("# Run 4 C1 QEMU rate table (#1213, #1211, #1223 QEMU side), generated by table.py from the files in this directory; RESULT per group below")
out.append("# boot banner: (table; each file's line 2 has its banner)")
out.append("# CPU 1 stopped = any `iso_latency_*: TIMEOUT` line, a stress-test timeout, a SLOTTRACE fault dump, or a panic on CPU 1. Litmus = test_litmus_patterns_pass [FAIL] (#1230, separate). 120 s cap: thread=single runs end at the cap before the suite finishes (the latency and stress tests run well before it).")
summary = []
for prefix, desc in groups:
    files = sorted(glob.glob(os.path.join(d, prefix + "*.txt")))
    if not files:
        continue
    out.append("")
    out.append(f"## {desc}: {prefix}NN.txt")
    out.append("| file | rc | samples | iso TIMEOUT lines | stress | dump | panic | lifecycle (#1211) | other [FAIL] | CPU 1 stopped |")
    out.append("|---|---|---|---|---|---|---|---|---|---|")
    n = stop = lit = lcf = lcn = cap = pds = 0
    for f in files:
        s = open(f, errors='replace').read()
        rc = g(r'qemu rc (\d+)', s)
        samples = g(r'Samples collected: (isolated=\d+/\d+, normal=\d+/\d+)', s)
        nto = len(re.findall(r'iso_latency_\w+: TIMEOUT after', s))
        st = g(r'slot_reuse_stress: (iterations=\d+ same_slot=\d+ ran=\d+ bad=\d+ first_bad_token=\d+ timeouts=\d+)', s)
        stid = g(r'slot_reuse_stress: iteration \d+: (task id=\d+ state=\d+)', s, '')
        dump = g(r'SLOTTRACE-DUMP-BEGIN why=(\S+)', s)
        panic = 'yes' if 'KERNEL PANIC' in s else '-'
        pmsg = g(r'(schedule returned!|stack overflow|terminated/destroyed task selected)', s, '')
        lc = g(r'\[C1-1211\] lifecycle (PASS|FAIL)', s)
        pdm = g(r'pdm: (migrations=\d+ moved=\d+ runs=\d+/\d+ bad=\d+)', s, '')
        if lc != '-':
            lcn += 1
            lcf += lc == 'FAIL'
        fails = sorted(set(re.findall(r'\[FAIL\] (test_\w+)', s)))
        lit += 'test_litmus_patterns_pass' in fails
        stopped = nto > 0 or 'timeouts=1' in st or dump not in ('-', 'lifecycle_leak_1211', 'pd_stuck') or panic == 'yes'
        n += 1
        stop += stopped
        cap += rc == '124'
        pds += dump == 'pd_stuck'
        out.append(f"| {os.path.basename(f)} | {rc} | {samples} | {nto} | {st} {stid} | {dump} | {panic} {pmsg} | {lc} | {' '.join(fails) or '-'} {pdm} | {'YES' if stopped else 'no'} |")
    line = f"{desc}: CPU 1 stopped {stop}/{n}; preempt_disabled stuck dump {pds}/{n}; 120 s cap {cap}/{n}; litmus [FAIL] {lit}/{n}"
    if lcn:
        line += f"; #1211 lifecycle FAIL {lcf}/{lcn}"
    out.append("")
    out.append("RESULT " + line)
    summary.append(line)
out.insert(3, "# Summary:")
for i, l in enumerate(summary):
    out.insert(4 + i, "#   " + l)
print("\n".join(out))
