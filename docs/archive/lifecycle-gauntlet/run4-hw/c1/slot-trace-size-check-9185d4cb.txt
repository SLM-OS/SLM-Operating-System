import re,sys,glob
def strip_calls(s, name):
    out=[];i=0
    pat=re.compile(r'\b'+name+r'\s*\(')
    while True:
        m=pat.search(s,i)
        if not m: out.append(s[i:]); break
        out.append(s[i:m.start()]); j=m.end(); d=1
        while d: 
            c=s[j]; d+= (c=='(') - (c==')'); j+=1
        # swallow trailing ';' for statement macros
        k=j
        while k<len(s) and s[k] in ' \t': k+=1
        if name!='SLOT_TRACE_PACK' and k<len(s) and s[k]==';': j=k+1
        out.append('' if name!='SLOT_TRACE_PACK' else '0')
        i=j
    return ''.join(out)
def strip_if(s):
    # remove #if defined(CONFIG_SLOT_TRACE) ... #endif blocks (no nesting inside assumed)
    return re.sub(r'#if defined\(CONFIG_SLOT_TRACE\)\n.*?#endif[^\n]*\n','',s,flags=re.S)
for f in sys.argv[1:]:
    s=open(f).read(); o=s
    for n in ['SLOT_TRACE_ONLY','SLOT_TRACE_RQ_LOCKED','SLOT_TRACE_RQ_UNLOCKING','slot_trace_pd_tick','SLOT_TRACE_PACK','SLOT_TRACE']:
        s=strip_calls(s,n)
    s=s.replace('#include "slot_trace.h"\n','')
    if not f.endswith('slot_trace.c'): s=strip_if(s)
    if s!=o: open(f,'w').write(s); print('stripped',f)
