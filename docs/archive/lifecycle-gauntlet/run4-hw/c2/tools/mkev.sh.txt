#!/bin/bash
# usage: mkev.sh <rundir> <label> <head> <elf> <cell> <outfile>
D=$1; L=$2; HEAD=$3; ELF=$4; CELL=$5; OUT=$6
E=$ELF; SHA=$(sha256sum $ELF | cut -c1-64)
M=$(cat $D/$L.meta)
RC=$(echo "$M" | grep -o "rc=[0-9]*" | head -1); RES=$(echo "$M" | grep -o "result=[^ ]*")
LOADS=$(echo "$M" | grep -o "load_start=\[[^]]*\]"); LOADE=$(echo "$M" | grep -o "load_end=\[[^]]*\]")
WALL=$(echo "$M" | grep -o "wall=[0-9.]*" | cut -c1-10)
FAILS=$(grep -F "[FAIL]" $D/$L.raw.log | grep -v "test(s) failed\|total failure\|tests had failures" | sed 's/^ *\[FAIL\] //' | paste -sd ';' -)
BANNER=$(grep -m1 "SLM-OS v" $D/$L.raw.log | sed 's/^ *//')
{
echo "# $CELL $L: make test WORK_STEALING=ON QEMU command (hand-run copy + qmp/gdb sockets), head $HEAD, elf sha256 $SHA, $LOADS $LOADE (1/5/15 min loadavg, 8 cores), $WALL s, $RC, RESULT $RES${FAILS:+, FAIL lines: $FAILS}"
echo "# boot banner: $BANNER"
echo "# meta: $M"
cat $D/$L.raw.log
if [ -s $D/$L.samples.txt ]; then
  echo
  echo "################ QMP PC samples (all vCPUs) and gdb task dump ################"
  echo "# symbolized PCs (aarch64-none-elf-addr2line -f against the elf above):"
  for a in $(grep PCSAMPLE $D/$L.samples.txt | grep -o "pc=[0-9a-f]*\|x30=[0-9a-f]*" | cut -d= -f2 | sort -u); do
    echo "#   0x$a $(aarch64-none-elf-addr2line -f -e $ELF 0x$a | head -1)"; done
  echo "# PC histogram (count label sample cpu pc x30):"
  grep PCSAMPLE $D/$L.samples.txt | awk '{print $2, $4, $5, $6}' | sort | uniq -c | sed 's/^/#  /'
  cat $D/$L.samples.txt
fi
} > $OUT
