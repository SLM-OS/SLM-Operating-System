#!/bin/bash
# Offline Jetson prep for C2 step 3, given a head.
# usage: jetson_prep.sh <head> [compare-with-head]
# Builds make kernel PLATFORM=JETSON_ORIN_NANO at <head> in the l4-c2
# worktree (clean), snapshots slmos.elf/.bin with sha256, checks the
# config and the embedded commit, compares the ELF with an earlier
# snapshot (function sizes and section layout), writes
# step3-procedure-<head8>.txt from the template, restores the worktree.
set -eu
HEAD=$1; CMP=${2:-9185d4cb}
J=$(cd $(dirname "$0") && pwd)
W=/home/john/projects/CS-496-Capstone-SLM-Operating-System/.claude/worktrees/l4-c2
EVD=${EVD_OVERRIDE:-/home/john/projects/CS-496-Capstone-SLM-Operating-System/.claude/worktrees/l4-ev-pi5-c2/docs/archive/lifecycle-gauntlet/run4-hw/c2}
[ -z "$(git -C $W status --porcelain)" ] || { echo "worktree dirty"; exit 1; }
git -C $W fetch -q origin
git -C $W checkout -q --detach "$HEAD"
FULL=$(git -C $W rev-parse HEAD); SHORT=${FULL:0:8}
trap 'git -C $W checkout -q c2/ws-1219' EXIT
make -C $W kernel-clean > /dev/null 2>&1
make -C $W kernel PLATFORM=JETSON_ORIN_NANO > $J/build-$SHORT.log 2>&1 || { tail -30 $J/build-$SHORT.log; exit 1; }
SNAP=${SNAP_SUFFIX:-}; mkdir -p $J/$SHORT$SNAP; cp $W/build/kernel/slmos.elf $W/build/kernel/slmos.bin $J/$SHORT$SNAP/
ELFSHA=$(sha256sum $J/$SHORT$SNAP/slmos.elf | cut -c1-64); BINSHA=$(sha256sum $J/$SHORT$SNAP/slmos.bin | cut -c1-64)
CFG=$(grep -E "^(ENABLE_WORK_STEALING|JETSON_HW_TICK|SECONDARY_PREEMPT):" $W/build/kernel/CMakeCache.txt | tr '\n' ' ')
NC=$(strings $J/$SHORT$SNAP/slmos.elf | grep -c "$FULL" || true)
echo "head $FULL elf $ELFSHA bin $BINSHA config: $CFG commit-string hits: $NC"
OUT=$EVD/jetson-elf-compare-$CMP-$SHORT.txt
if [ -e $J/$CMP/slmos.elf ]; then
  A=$J/$CMP/slmos.elf; B=$J/$SHORT$SNAP/slmos.elf
  diff <(aarch64-none-elf-nm -S $A | awk '$3 ~ /[tT]/ {print $4, $2}' | sort) \
       <(aarch64-none-elf-nm -S $B | awk '$3 ~ /[tT]/ {print $4, $2}' | sort) \
       | grep "^[<>]" | awk '{print $2}' | sort -u > $J/fn-diff-$CMP-$SHORT.txt || true
  {
  echo "# Jetson production ELF compare: $CMP (sha256 $(sha256sum $A | cut -c1-64)) vs $SHORT (sha256 $ELFSHA), RESULT $(wc -l < $J/fn-diff-$CMP-$SHORT.txt) functions changed size or appeared/disappeared"
  echo "# boot banner: n/a (static compare)"
  echo "## git diff --stat $CMP $SHORT"; git -C $W diff --stat $CMP $SHORT
  echo "## functions whose size differs (or exist in one only), with source file"
  for f in $(cat $J/fn-diff-$CMP-$SHORT.txt); do
    a=$(aarch64-none-elf-nm -S $A | awk -v f=$f '$4==f{print $2; exit}'); b=$(aarch64-none-elf-nm -S $B | awk -v f=$f '$4==f{print $2; exit}')
    ad=$(aarch64-none-elf-nm $B | awk -v f=$f '$3==f{print "0x"$1; exit}'); [ -z "$ad" ] && ad=$(aarch64-none-elf-nm $A | awk -v f=$f '$3==f{print "0x"$1; exit}')
    src=$(aarch64-none-elf-addr2line -e $B $ad 2>/dev/null | sed 's|.*/kernel/|kernel/|')
    echo "$f: ${a:-absent} -> ${b:-absent} ($src)"
  done
  echo "## section headers $CMP"; aarch64-none-elf-readelf -SW $A | grep "^\s*\["
  echo "## section headers $SHORT"; aarch64-none-elf-readelf -SW $B | grep "^\s*\["
  } > $OUT
  echo "compare -> $OUT"; sed -n "1p" $OUT; sed -n "/^## functions/,/^## section/p" $OUT | head -40 || true
fi
sed -e "s/@FULL@/$FULL/g; s/@FULL12@/${FULL:0:12}/g; s/@ELFSHA@/$ELFSHA/; s/@BINSHA@/$BINSHA/; s/@SHORT@/$SHORT/g" $J/procedure.tmpl > $EVD/step3-procedure-$SHORT.txt
sed -i "2a # generated by tools/v2/jetson_prep.sh.txt; config: $CFG" $EVD/step3-procedure-$SHORT.txt
cp $J/build-$SHORT.log $EVD/jetson-build-$SHORT.log.txt
for f in jetson_prep.sh procedure.tmpl j_stage.sh j_kexec.sh j_esp.sh; do cp $J/$f $EVD/tools/v2/$f.txt; done
echo "procedure -> $EVD/step3-procedure-$SHORT.txt"
