#!/bin/bash
# Unattended C2 cell: build a head's test kernel once (clean), then N runs
# of the make-test QEMU command with QMP/gdb sampling, evidence files and
# a summary table. One QEMU at a time.
# usage: run_cell.sh <head> <ON|OFF> <cell-name> [n=20]
set -u
HEAD=$1; WS=$2; CELL=$3; N=${4:-20}
S=$(cd $(dirname "$0") && pwd)
W=/home/john/projects/CS-496-Capstone-SLM-Operating-System/.claude/worktrees/l4-c2
EVD=/home/john/projects/CS-496-Capstone-SLM-Operating-System/.claude/worktrees/l4-ev-pi5-c2/docs/archive/lifecycle-gauntlet/run4-hw/c2
RD=$S/runs/$CELL; mkdir -p $RD $S/elf
log() { echo "[$(date -u +%H:%M:%S)] $*" | tee -a $RD/cell.log; }
if [ -n "$(git -C $W status --porcelain)" ]; then log "worktree dirty, abort"; exit 1; fi
git -C $W fetch -q origin 2>/dev/null
git -C $W checkout -q --detach "$HEAD" || { log "checkout $HEAD failed"; exit 1; }
FULL=$(git -C $W rev-parse HEAD); SHORT=${FULL:0:8}
log "cell $CELL head $FULL WORK_STEALING=$WS n=$N build-load $(cut -d' ' -f1-3 /proc/loadavg)"
make -C $W kernel-test-clean > $RD/build.log 2>&1
make -C $W kernel-test WORK_STEALING=$WS >> $RD/build.log 2>&1 || { log "build failed, see build.log"; exit 1; }
got=$(grep "^ENABLE_WORK_STEALING:BOOL=" $W/build/kernel-test/CMakeCache.txt | cut -d= -f2)
[ "$got" = "$WS" ] || { log "CMakeCache ENABLE_WORK_STEALING=$got, wanted $WS"; exit 1; }
ELF=$S/elf/$CELL.elf; cp $W/build/kernel-test/slmos.elf $ELF
export C2_OFFS=$S/elf/$CELL.offs.json
$S/mkoffs.sh $W $W/build/kernel-test $C2_OFFS > /dev/null || { log "offsets failed"; exit 1; }
log "elf sha256 $(sha256sum $ELF | cut -c1-64), CMakeCache ENABLE_WORK_STEALING:BOOL=$got"
for i in $(seq -w 1 $N); do
  [ -e $RD/run$i.meta ] && continue
  $S/run_one.sh $ELF $RD run$i >> $RD/cell.log 2>&1
  log "run$i $(cut -d' ' -f2-3 $RD/run$i.meta) class=$($S/classify.sh $RD run$i $ELF)"
done
mkdir -p $EVD
DESC="cell $CELL (head $SHORT, WORK_STEALING=$WS, 1 QEMU at a time)"
T=$EVD/$CELL-summary.txt
{
echo "# $CELL summary: make test QEMU command, head $FULL, WORK_STEALING=$WS (CMakeCache $got), TCG thread=multi (no -accel flag), elf sha256 $(sha256sum $ELF | cut -c1-64), N=$N; classes A=#1219 batch stress, B=panic PC (#1213-like), C=test_litmus_patterns_pass"
echo "# boot banner: $(grep -m1 'SLM-OS v' $RD/run01.raw.log | sed 's/^ *//')"
echo
echo "| File | load_start (1/5/15) | rc | Class |"
echo "|---|---|---|---|"
for i in $(seq -w 1 $N); do
  $S/mkev.sh $RD run$i $SHORT $ELF "$DESC" $EVD/$CELL-run$i.txt
  sed -i "1s/make test WORK_STEALING=ON QEMU command/make test WORK_STEALING=$WS QEMU command/" $EVD/$CELL-run$i.txt
  m=$(cat $RD/run$i.meta)
  echo "| $CELL-run$i.txt | $(echo "$m" | grep -o 'load_start=\[[^]]*\]' | sed 's/load_start=\[\(.*\)\]/\1/') | $(echo "$m" | grep -o 'rc=[0-9]*' | head -1 | cut -d= -f2) | $($S/classify.sh $RD run$i $ELF) |"
done
echo
for k in A B C; do echo "Class $k: $(for i in $(seq -w 1 $N); do $S/classify.sh $RD run$i $ELF; done | grep -c "\b$k\b") of $N"; done
echo "Any failure: $(for i in $(seq -w 1 $N); do $S/classify.sh $RD run$i $ELF; done | grep -vc '^PASS$') of $N"
echo "Load (1-min) at run start: min $(cat $RD/run*.meta | grep -o 'load_start=\[[0-9.]*' | cut -d[ -f2 | sort -n | head -1), max $(cat $RD/run*.meta | grep -o 'load_start=\[[0-9.]*' | cut -d[ -f2 | sort -n | tail -1)"
} > $T
mkdir -p $EVD/tools/v2; for f in run_cell.sh run_one.sh qmp_sample.py tasks.py mkoffs.sh offs.c classify.sh mkev.sh; do cp $S/$f $EVD/tools/v2/$f.txt; done
cp $C2_OFFS $EVD/$CELL-offs.json
git -C $W checkout -q c2/ws-1219
log "cell $CELL done; table $T"
tail -6 $T | tee -a $RD/cell.log
