+ * slot_trace.h - Per-CPU task-slot lifecycle trace (#1213)
+ *
+ * Debug-only. Built with `make ... SLOT_TRACE=ON` (CMake -DSLOT_TRACE=ON,
+ * which defines CONFIG_SLOT_TRACE). With the option off every macro below
+ * expands to nothing and slot_trace.c compiles to an empty unit, so the
+ * kernel's .text is unchanged.
+ *
+ * What it records, per task_table slot: task_create (slot claimed, context
+ * written), task_destroy (entry and the generation bump), task_exit,
+ * scheduler_terminate_task, run-queue inserts, zombie set and reap,
+ * switch_to entry (outgoing and incoming slot, generation, the incoming
+ * context's saved sp/x19 as read just before the switch), switch_to return,
+ * a task's first instruction in C (task_entry_trampoline), and test markers.
+ * Every entry carries the CPU, CNTVCT, a global sequence number (one relaxed
+ * atomic increment, for a total order across CPUs) and the four run-queue
+ * lock holders at the time of the record.
+ *
+ * Writers: ring c is written only by CPU c, with local IRQs masked for the
+ * few stores of one entry. No locks and no prints on the record path.
+ *
+ * Dump: the first EL1 synchronous exception (including the bad-SP case,
+ * which the el1_sync vector moves onto a per-CPU emergency stack before
+ * anything touches memory) or the first panic freezes every ring and prints
+ * all of them, then exits QEMU through semihosting.
+ */

+/*
+ * slot_trace.c - Per-CPU task-slot lifecycle trace (#1213). See slot_trace.h.
+ *
+ * Compiles to an empty unit unless CONFIG_SLOT_TRACE is defined.
+ */
+
+#include "slot_trace.h"
+
+#if defined(CONFIG_SLOT_TRACE)
+
+#include "config.h"
+#include "task.h"
+#include "sched.h"
+#include "smp.h"
+#include "uart.h"
+#include "semihosting.h"
+#include <stddef.h>
+
+#define SLOT_TRACE_LEN 512U     /* entries per CPU (power of two) */
+
+struct slot_trace_ent {
+    uint64_t ts;        /* CNTVCT_EL0 */
+    uint32_t seq;       /* global order */
+    uint16_t ev;
+    uint8_t  cpu;
+    uint8_t  pad0;
+    int16_t  slot;      /* task_table index, -1 if not a table task */
+    uint16_t state;
+    uint32_t gen;
+    uint32_t id;
+    uint32_t holders;   /* rq lock holder per CPU, 4 bits each (0 = free, n+1 = CPU n) */
+    uint64_t a, b, c;
+};
+
+static struct slot_trace_ent slot_trace_ring[MAX_CPUS][SLOT_TRACE_LEN];
+static volatile uint32_t slot_trace_idx[MAX_CPUS];
+static volatile uint32_t slot_trace_seq;
+static volatile uint32_t slot_trace_frozen;
+volatile uint8_t slot_trace_rq_holder[MAX_CPUS];
+
+/* Emergency stacks for the bad-SP vector path (vectors.S). */
+#define SLOT_TRACE_EMERG_STACK 16384
+__attribute__((aligned(16)))
+uint8_t slot_trace_emerg[4][SLOT_TRACE_EMERG_STACK];
+
+static inline uint64_t st_cntvct(void)
+{
+    uint64_t v;
+    __asm__ volatile("mrs %0, cntvct_el0" : "=r"(v));
+    return v;
+}
+
+static int st_slot(const struct task *t)
+{
+    const struct task *base = task_slot(0);
+    if (!t || !base) {
+        return -1;
+    }
+    if (t < base || t >= base + MAX_TASKS) {
+        return -1;
+    }
+    return (int)(t - base);
+}
+
+uint64_t slot_trace_pack(const struct task *t)
+{
+    if (!t) {
+        return 0xFFFFULL << 48;
+    }
+    int s = st_slot(t);
+    return ((uint64_t)(uint16_t)(int16_t)s << 48) |
+           ((uint64_t)t->generation << 16) |
+           (uint64_t)((uint16_t)t->state);
+}
+
+void slot_trace_rec(uint32_t ev, const struct task *t,
+                    uint64_t a, uint64_t b, uint64_t c)
+{
+    if (__atomic_load_n(&slot_trace_frozen, __ATOMIC_RELAXED)) {
+        return;
+    }
+    uint32_t cpu = cpu_id();
+    if (cpu >= MAX_CPUS) {
+        return;
+    }
+    uint64_t daif;
+    __asm__ volatile("mrs %0, daif\n\tmsr daifset, #3" : "=r"(daif) :: "memory");
+
+    uint32_t i = slot_trace_idx[cpu];
+    struct slot_trace_ent *e = &slot_trace_ring[cpu][i & (SLOT_TRACE_LEN - 1)];
+    e->ts = st_cntvct();
+    e->seq = __atomic_fetch_add(&slot_trace_seq, 1, __ATOMIC_RELAXED);
+    e->ev = (uint16_t)ev;
+    e->cpu = (uint8_t)cpu;
+    e->slot = (int16_t)st_slot(t);
+    e->state = t ? (uint16_t)t->state : 0xFFFF;
+    e->gen = t ? t->generation : 0;
+    e->id = t ? t->id : 0;
+    uint32_t h = 0;
+    for (uint32_t k = 0; k < 4 && k < MAX_CPUS; k++) {
+        h |= ((uint32_t)slot_trace_rq_holder[k] & 0xF) << (4 * k);
+    }
+    e->holders = h;
+    e->a = a;
+    e->b = b;
+    e->c = c;
+    slot_trace_idx[cpu] = i + 1;
+
+    __asm__ volatile("msr daif, %0" :: "r"(daif) : "memory");
+}
+
+static const char *st_name(uint32_t ev)
+{
+    switch (ev) {
+    case ST_CREATE_ALLOC:    return "CREATE_ALLOC";
+    case ST_CREATE_CTX_DONE: return "CREATE_CTX_DONE";
+    case ST_DESTROY:         return "DESTROY";

docs/archive/lifecycle-gauntlet/run4-hw/c1/qemu-1213-trace-multi-run04/05/06

 2. CPU 0 (test) sees TERMINATED and task_destroy()s the slot (gen g -> g+1)
    while CPU 1 is still in schedule() for it.
 3. CPU 1 parks the slot as rq->zombie and switches to idle; idle's
    schedule() reaps the zombie with a second task_destroy().
 4. CPU 0's next task_create() claims the same slot (id set under
    task_lock, state still the old TERMINATED until after the unlock).
 5. The reaper's TERMINATED check passes; it clears the new task's id,
    stack_base and stack_top (gen g+1 -> g+2).
 6. task_create copies the cleared stack_top into context.sp (read back
    as 0), queues the task on CPU 1; CPU 1 switches in with SP 0 and takes
    the EL1 translation fault at the first stack store in
    task_entry_trampoline (ESR 0x96000045, FAR = SP - 0x30).
Run 3's hypothesis (switch_to reading a half-written context) is not what
the traces show: the context was complete, its sp field held 0 because
the second destroyer had cleared stack_top.

Fix:
 - task->on_cpu (the former padding word after generation): set to 1 by
   the CPU that switches the task in, under its rq_lock; cleared by the
   same CPU with a store-release immediately after switch_to returns on
   the next task's stack (sched_finish_switch). switch_to now returns its
   `old` argument (x0 / %rdi are untouched by the restore half), and
   task_entry_wrapper passes it to task_entry_trampoline, which calls
   sched_finish_switch first. task_destroy reads state, then on_cpu, with
   load-acquire and waits for 0 (2 s bound; on timeout it warns and leaks
   the slot rather than free a stack in use).
 - The zombie reaper records zombie_gen = current->generation when it
   parks the exiting task (on_cpu is still 1, so no destroy has run) and
   reclaims through task_destroy_gen(), which does nothing if the
   generation has changed: the owner destroyed the task first and the
   slot may already be a new task.
 - task_create / task_alloc set state = TASK_READY (and on_cpu = 0) in
   the task_lock section that claims the slot, so a stale destroyer can
   no longer read the previous occupant's TERMINATED on a claimed slot.

Ordering argument (Arm ARM DDI 0487, B2.3 "Definition of the Arm memory
model": Barrier-ordered-before, Observed-by, Ordered-before; C6.2 STLR,
LDAPR, LDAXR):
 - On the exiting CPU, the stores of switch_to's save half (STP/STR of
   x19-x30, SP, q0-q31, FPCR, FPSR, DAIF into prev->context) and every
   access to prev's stack precede, in program order, the STLR of
   prev->on_cpu = 0. A Store-Release is Barrier-ordered-after every
   explicit memory effect before it in program order.
 - The destroyer's on_cpu load is LDAPR (GCC's __ATOMIC_ACQUIRE with
   FEAT_LRCPC); a Load-AcquirePC is Barrier-ordered-before every memory
   effect after it in program order: the task_lock acquire, the slot
   clear, the generation bump, pmm_free_pages of the stack and, later,
   task_create's writes to the slot and to the reused stack pages.
 - When that load reads 0 from the STLR, the STLR is Observed-by it, so
   by Ordered-before transitivity the exiting CPU's context save and
   stack use are ordered before the destroyer's writes: no late save can
   overwrite a new occupant's context, and no freed stack is in use.
 - The destroyer reads state (LDAPR) before on_cpu, so a TERMINATED it
   observes comes with the on_cpu = 1 of the task's switch-in: that store
   precedes, on the CPU running the task, the STLR of the rq_lock release
   in schedule(), which is Barrier-ordered-before the LDAXR of the
   rq_lock acquire in scheduler_terminate_task (Store-Release then
   Load-Acquire in program order), which precedes the TERMINATED store.

+### Task Slot Reclaim (#1213)
+
+A terminated task can be reclaimed by two parties: its owner (code that
+waited for `TASK_TERMINATED` and calls `task_destroy`) and the run queue's
+zombie reaper (the next `schedule()` on the CPU it exited on). Three rules
+keep a slot from being freed or reused under a CPU that still uses it:
+
+- **`task->on_cpu`.** The CPU that switches a task in sets it to 1 (under
+  its `rq_lock`). The same CPU clears it with a store-release right after
+  `switch_to` returns on the next task's stack (`sched_finish_switch`, also
+  called first thing in `task_entry_trampoline` for a new task), that is,
+  after the outgoing task's context has been saved and its stack left.
+  `task_destroy` reads it with a load-acquire and waits for 0 (2 s bound,
+  then it warns and leaks the slot) before it frees the stack or clears
+  the slot. `TASK_TERMINATED` is published before the switch-out, so an
+  owner that sees it can be inside that window.
+- **Zombie generation.** `schedule()` records `zombie_gen =
+  current->generation` when it parks an exiting task as `rq->zombie`
+  (`on_cpu` is still 1, so no destroy can have run yet), and the reaper
+  calls `task_destroy_gen(zombie, zombie_gen)`, which reclaims only if the
+  slot still has that generation. If the owner destroyed the task first,
+  the generation has moved on and the slot, possibly already a new task,
+  is left alone.
+- **Claim publishes state.** `task_create` / `task_alloc` set
+  `state = TASK_READY` (and `on_cpu = 0`) in the same `task_lock` section
+  that claims the slot, so a stale destroyer can no longer see the previous
+  occupant's `TASK_TERMINATED` on a slot that already has a new id.
+
+Before these rules the reaper's second `task_destroy` raced the owner's
+next `task_create` on the same slot: it passed the TERMINATED check, cleared
+the new task's id and stack pointers mid-construction, and the new task ran
+with a saved SP of 0 (first-fault traces in
+`docs/archive/lifecycle-gauntlet/run4-hw/c1/` on `gauntlet-4-hw-pi5`). The
+debug build option `SLOT_TRACE=ON` (`kernel/include/slot_trace.h`) records
+these lifecycle events per CPU and dumps them on the first EL1 fault or
+panic.
+
 **Lock strategy:**
 - Single-queue operations (add, remove, schedule) use only that CPU's lock
 - Cross-queue operations (migration) lock both queues in CPU ID order to prevent deadlock

+**Task slot reclaim: `on_cpu` and the zombie generation (#1213):** `TASK_TERMINATED` is published before the exiting CPU switches out, so an owner that sees it and calls `task_destroy` can be inside that CPU's `schedule()`/`switch_to()`. `task_destroy` therefore waits (load-acquire, 2 s bound) for `task->on_cpu == 0`, which the CPU that ran the task stores with release semantics after `switch_to` returns on the next task's stack (`sched_finish_switch`; `switch_to` returns the task it switched away from, and `task_entry_wrapper` passes it to `task_entry_trampoline`).
