import re,sys
# Round 2 selftest logs: CPUs 1 and 2 dump concurrently (each must be one
# contiguous block ending in "System halted." and the end marker), then CPU 3
# stops 12 lines into its dump. Count unprefixed lines inside the dump region.
for p in sys.argv[1:]:
    L=open(p,errors='replace').read().replace('\r','').split('\n')
    blocks=[]; cur=None
    for l in L:
        m=re.match(r'^\[cpu(\d+)\] ',l); c=m.group(1) if m else None
        if c!=cur:
            if c is not None: blocks.append([c,0])
            cur=c
        if c is not None: blocks[-1][1]+=1
    seq=[b[0] for b in blocks]
    halted=sum(1 for l in L if re.match(r'^\[cpu[12]\] System halted\.$',l))
    ends=sum(1 for l in L if re.match(r'^\[cpu[12]\] fault_dump: end, releasing the dump lock$',l))
    ok = seq[:2] in (['1','2'],['2','1']) and len(seq)==3 and seq[2]=='3' and halted==2 and ends==2
    print(p.split('/')[-1],'blocks=',[tuple(b) for b in blocks],'halted=',halted,'ends=',ends,'OK' if ok else 'BAD')
