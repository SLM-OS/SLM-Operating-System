import re,sys
# Per log: lines prefixed [cpuN] must form contiguous blocks, one block per CPU,
# and no unprefixed line may fall inside a block.
for p in sys.argv[1:]:
    L=open(p,errors='replace').read().replace('\r','').split('\n')
    blocks=[]; cur=None; stray=0
    for l in L:
        m=re.match(r'^\[cpu(\d+)\] ',l)
        c=m.group(1) if m else None
        if c!=cur:
            if c is not None: blocks.append([c,0])
            cur=c
        if c is not None: blocks[-1][1]+=1
    seq=[b[0] for b in blocks]
    ok = len(seq)==len(set(seq))==2
    halted=sum(1 for l in L if re.match(r'^\[cpu\d+\] System halted\.$',l))
    done=[l for l in L if 'dumps completed' in l]
    print(p.split('/')[-1], 'blocks=',[(b[0],b[1]) for b in blocks],'halted_lines=',halted, done[0] if done else 'NO-DONE', 'OK' if ok and halted==2 else 'BAD')
