#!/bin/bash
# savebatch.sh <labctl dir name> <dest prefix> <first boot number> <n> <trigger log name> <title...>
# Saves run_01..run_n of one boot_test call as <prefix>-bootNN.txt, plus the trigger log.
S=/tmp/claude-1000/-home-john-projects-CS-496-Capstone-SLM-Operating-System/332cf057-6ae0-47f7-8ea7-10f13477578f/scratchpad
E=/home/john/projects/CS-496-Capstone-SLM-Operating-System/.claude/worktrees/Issues-1262-and-1244/.wt/evj/docs/archive/switch-gauntlet/run5-hw/1262-1244
dir=$1; prefix=$2; first=$3; n=$4; trig=$5; shift 5
for i in $(seq 1 $n); do
  b=$(printf %02d $((first + i - 1)))
  bash $S/save.sh /var/lib/labctl/output/$dir/run_$(printf %02d $i).txt $E/$prefix-boot$b.txt "$*, boot $b"
done
[ -f "$S/$trig" ] && cp "$S/$trig" $E/ && cat "$S/$trig"
grep -a -c "Cannot migrate running" /var/lib/labctl/output/$dir/run_*.txt
