def sub(path, old, new, count=1):
    p = R + '/' + path
    s = open(p).read()

    open(p, 'w').write(s.replace(old, new))

# ---- pmm: helpers --------------------------------------------------------------
HELPERS = '''
/*
 * Page-aligned range that holds the DTB blob (#1244): down to the page of
 * its first byte, up to the page after its last. Pure.
 */
bool pmm_dtb_reserve_range(uint64_t blob_addr, uint64_t blob_size,
                           dtb_memreserve_t *out)
{
    if (!out || blob_size == 0 || blob_addr + blob_size < blob_addr) {
        return false;
    }
    uint64_t first = blob_addr & ~(uint64_t)(PAGE_SIZE - 1);
    uint64_t last = (blob_addr + blob_size + PAGE_SIZE - 1) &
                    ~(uint64_t)(PAGE_SIZE - 1);
    out->addr = first;
    out->size = last - first;
    return true;
}

/*
 * Number of free blocks that overlap [start, end); the first one found is
 * returned through first_addr/first_order (may be NULL). Walks every free
 * list: boot-time and test use only. Caller holds pmm_lock or runs before
 * any other CPU allocates.
 */
size_t pmm_free_overlap_count(uintptr_t start, uintptr_t end,
                              uintptr_t *first_addr, unsigned int *first_order)
{
    size_t n = 0;
    for (unsigned int o = 0; o <= MAX_ORDER; o++) {
        size_t bsize = order_to_size(o);
        for (struct free_block *b = buddy_state.free_lists[o]; b; b = b->next) {
            uintptr_t a = (uintptr_t)b;
            if (a < end && a + bsize > start) {
                if (n == 0) {
                    if (first_addr) {
                        *first_addr = a;
                    }
                    if (first_order) {
                        *first_order = o;
                    }
                }
                n++;
            }
        }
    }
    return n;
}

size_t pmm_free_overlap_count_locked(uintptr_t start, uintptr_t end)
{
    irq_flags_t flags = spin_lock_irqsave(&pmm_lock);
    size_t n = pmm_free_overlap_count(start, end, NULL, NULL);
    spin_unlock_irqrestore(&pmm_lock, flags);
    return n;
}
'''

if MODE == 'full':

/*
 * Every range pmm_add_region_split carves out of a region: the firmware
 * /memreserve/ list, the user reserves, and the DTB blob itself (#1244).
 * Writes at most `max` entries; returns the number written.
 */
int pmm_collect_reserves(dtb_memreserve_t *out, int max)
{
    int n_rsv = dtb_get_memreserves(out, max < DTB_MAX_MEMRESERVES
                                             ? max : DTB_MAX_MEMRESERVES);
    /* User-registered reserves (e.g. GPU kexec handoff state — see
     * `pmm_user_reserve_add`). The carve helper handles unsorted,
     * overlapping, and out-of-region entries identically to firmware
     * reserves, so the lists can be concatenated without further
     * bookkeeping. */
    for (size_t i = 0; i < pmm_n_user_reserves && n_rsv < max; i++) {
        out[n_rsv].addr = pmm_user_reserves[i].addr;
        out[n_rsv].size = pmm_user_reserves[i].size;
        n_rsv++;
    }
    /* The DTB blob (#1244). Its own reserve map does not cover it, and
     * the loader decides where it sits: kexec-tools puts it in the first
     * free pages after the kernel image, which is where the heap starts;
     * Pi 5 firmware puts it in low RAM; UEFI leaves it above the heap.
     * Reserved from the blob's actual address and totalsize, so every
     * boot path is covered whatever the loader does. */
    uint64_t blob_addr, blob_size;
    if (n_rsv < max && dtb_get_blob_range(&blob_addr, &blob_size) &&
        pmm_dtb_reserve_range(blob_addr, blob_size, &out[n_rsv])) {
        n_rsv++;
    }
    return n_rsv;
}
'''

# ---- pmm: boot-time check ------------------------------------------------------
sub('kernel/mm/pmm.c',
'''    buddy_state.initialized = true;

    INFO("PMM initialized (buddy allocator)");''',
'''    /*
     * #1244: no free block may overlap the DTB blob. The blob is read for
     * the whole life of the kernel (FDT lookups by drivers), and a free
     * block inside it has its list pointers written over the blob and its
     * pages handed out. Warn, do not panic: the kernel has run this way,
     * and the line is the evidence that names the cause.
     */
    {
        uint64_t blob_addr, blob_size;
        dtb_memreserve_t r;
        if (dtb_get_blob_range(&blob_addr, &blob_size) &&
            pmm_dtb_reserve_range(blob_addr, blob_size, &r)) {
            uintptr_t first = 0;
            unsigned int first_order = 0;
            size_t n = pmm_free_overlap_count((uintptr_t)r.addr,
                                              (uintptr_t)(r.addr + r.size),
                                              &first, &first_order);
            if (n) {
                WARN("PMM: %u free block(s) overlap the DTB at 0x%lx-0x%lx "
                     "(%u bytes; first block 0x%lx order %u; heap starts "
                     "0x%lx) (#1244)",
                     (unsigned)n, (unsigned long)r.addr,
                     (unsigned long)(r.addr + r.size), (unsigned)blob_size,
                     (unsigned long)first, first_order,
                     (unsigned long)buddy_state.heap_start);
            } else {
                INFO("PMM: DTB at 0x%lx-0x%lx (%u bytes) is clear of every "
                     "free block (heap starts 0x%lx)",
                     (unsigned long)r.addr, (unsigned long)(r.addr + r.size),
                     (unsigned)blob_size,
                     (unsigned long)buddy_state.heap_start);
            }
        }
    }

    buddy_state.initialized = true;

    INFO("PMM initialized (buddy allocator)");''')

# ---- tests ---------------------------------------------------------------------
TESTS = '''
/* ============================================================================
 * DTB blob reservation (#1244)
 *
 * The PMM carves the DTB blob's own range out of every region and checks at
 * init that no free block overlaps it. On Jetson kexec the loader puts the
 * blob in the first free pages after the kernel image, where the heap
 * starts; on Pi 5 the firmware puts it in low RAM, also inside the heap.
 * ============================================================================ */

static void test_dtb_reserve_range_page_aligns(void)
{
    dtb_memreserve_t r = { 0, 0 };
    /* Pi 5 style: unaligned start, unaligned size. */
    TEST_ASSERT_TRUE(pmm_dtb_reserve_range(0x2eff2e00ULL, 0x13a5c, &r));
    TEST_ASSERT_EQUAL_HEX64(0x2eff2000ULL, r.addr);
    TEST_ASSERT_EQUAL_HEX64(0x2f007000ULL - 0x2eff2000ULL, r.size);
    /* Already aligned: unchanged. */
    TEST_ASSERT_TRUE(pmm_dtb_reserve_range(0x80a9b000ULL, 0x2000, &r));
    TEST_ASSERT_EQUAL_HEX64(0x80a9b000ULL, r.addr);
    TEST_ASSERT_EQUAL_HEX64(0x2000ULL, r.size);
    /* One byte takes one page. */
    TEST_ASSERT_TRUE(pmm_dtb_reserve_range(0x1fffULL, 1, &r));
    TEST_ASSERT_EQUAL_HEX64(0x1000ULL, r.addr);
    TEST_ASSERT_EQUAL_HEX64(0x1000ULL, r.size);
    /* Rejected: empty, wrapping, no output. */
    TEST_ASSERT_FALSE(pmm_dtb_reserve_range(0x1000, 0, &r));
    TEST_ASSERT_FALSE(pmm_dtb_reserve_range(~0ULL - 4, 16, &r));
    TEST_ASSERT_FALSE(pmm_dtb_reserve_range(0x1000, 16, NULL));
}

/* The kexec layout of #1244: blob starts below the (64 KB aligned) end of
 * the image and runs about 950 KB into the heap. */
static void test_carve_dtb_straddling_heap_start(void)
{
    dtb_memreserve_t r;
    TEST_ASSERT_TRUE(pmm_dtb_reserve_range(0x80a9a000ULL, 0xe8000, &r));
    uintptr_t s[4], e[4];
    int n = pmm_carve_reserves(0x80aa0000UL, 0xa1000000UL, &r, 1, s, e, 4);
    TEST_ASSERT_EQUAL_INT(1, n);
    TEST_ASSERT_EQUAL_HEX64(0x80b82000UL, s[0]);     /* first page after the blob */
    TEST_ASSERT_EQUAL_HEX64(0xa1000000UL, e[0]);
}

/* The list pmm_init carves holds the live blob's range. */
static void test_collect_reserves_includes_live_dtb(void)
{
    uint64_t addr, size;
    if (!dtb_get_blob_range(&addr, &size)) {
        TEST_IGNORE_MESSAGE("no DTB blob on this platform or boot path");
    }
    dtb_memreserve_t want;
    TEST_ASSERT_TRUE(pmm_dtb_reserve_range(addr, size, &want));

    static dtb_memreserve_t list[DTB_MAX_MEMRESERVES + 2048 + 1];
    int n = pmm_collect_reserves(list, (int)(sizeof(list) / sizeof(list[0])));
    bool found = false;
    for (int i = 0; i < n; i++) {
        if (list[i].addr == want.addr && list[i].size == want.size) {
            found = true;
        }
    }
    TEST_ASSERT_TRUE_MESSAGE(found, "DTB blob range not in the PMM reserve list");
    TEST_ASSERT_TRUE(want.addr <= addr && want.addr + want.size >= addr + size);
}

/* No free block overlaps the live blob (fails on a kernel without the
 * reservation wherever the loader put the blob inside the heap: Jetson
 * kexec, Pi 5). */
static void test_no_free_block_overlaps_live_dtb(void)
{
    uint64_t addr, size;
    if (!dtb_get_blob_range(&addr, &size)) {
        TEST_IGNORE_MESSAGE("no DTB blob on this platform or boot path");
    }
    dtb_memreserve_t r;
    TEST_ASSERT_TRUE(pmm_dtb_reserve_range(addr, size, &r));
    TEST_ASSERT_EQUAL_UINT64(0, pmm_free_overlap_count_locked(
                                    (uintptr_t)r.addr,
                                    (uintptr_t)(r.addr + r.size)));
}

/* The overlap walk sees a free block and stops seeing it once allocated. */
static void test_free_overlap_count_sees_a_free_page(void)
{
    void *p = pmm_alloc_page();
    TEST_ASSERT_NOT_NULL(p);
    uintptr_t a = (uintptr_t)p;
    TEST_ASSERT_EQUAL_UINT64(0, pmm_free_overlap_count_locked(a, a + PAGE_SIZE));
    pmm_free_page(p);
    TEST_ASSERT_TRUE(pmm_free_overlap_count_locked(a, a + PAGE_SIZE) >= 1);
}
'''
