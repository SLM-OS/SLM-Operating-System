# output) with save.sh (both-ends check), copies the trigger log next to it, and prints
# the per-file counts with pdmsum.sh.

mkdir -p $E
dir=$1; dst=$2; trig=$3; shift 3
bash $S/save.sh /var/lib/labctl/output/$dir/run_01.txt $E/$dst "$*"

bash $S/pdmsum.sh $E/$dst
