#!/bin/bash
# ci.sh : the /ci skill's steps, in order, in .wt/ft (cd1557fe). One log per step in $L.
# Stops at the first failing step. Session #1284.
W=/home/john/projects/CS-496-Capstone-SLM-Operating-System/.claude/worktrees/Decide-1284/.wt/ft
L=/tmp/claude-1000/-home-john-projects-CS-496-Capstone-SLM-Operating-System/f616ae46-0930-4d47-ae76-ce0c02faf621/scratchpad/a/ci
export PATH=$HOME/.cargo/bin:$PATH
mkdir -p $L
cd $W || exit 1
echo "head $(git rev-parse --short=8 HEAD) dirty=$(git status --porcelain --untracked-files=no | wc -l) start $(date -u +%T)" > $L/summary.txt
step() {
  n=$1; name=$2; shift 2
  echo "$(date -u +%T) step $n $name: $*" >> $L/summary.txt
  bash -c "$*" > $L/step$n-$name.txt 2>&1
  rc=$?
  echo "$(date -u +%T) step $n $name rc=$rc" >> $L/summary.txt
  [ $rc -eq 0 ] || { echo "STOPPED at step $n" >> $L/summary.txt; exit $rc; }
}
step 01 build-arm64   "make kernel-clean && make kernel PLATFORM=QEMU_VIRT"
step 02 build-x86     "make kernel-clean && make kernel PLATFORM=X86_64"
step 03 build-jetson-real "make kernel-clean && make kernel PLATFORM=JETSON_ORIN_NANO AI_WEIGHTS=real && make kernel-test-clean && make kernel-test PLATFORM=JETSON_ORIN_NANO AI_WEIGHTS=real"
step 04 test-arm64    "make test"
step 05 test-x86      "make test PLATFORM=X86_64"
step 06 test-ai-sched "make test AI_SCHED=ON"
step 07 runtime-test  "make runtime-test"
step 08 braces        "make test-check-braces && make check-braces"
step 09 build-order   "make test-build-order"
step 10 jetson-layout "make test-jetson-uefi-layout"
step 11 diff-check    "git diff origin/main...HEAD --stat; git status --porcelain"
echo "ALL DONE $(date -u +%T)" >> $L/summary.txt
