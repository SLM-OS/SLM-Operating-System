import sys
# DEMO ONLY (not committed): leaves a cleared order-0 link in place, waits for the
# timer-tick walk, then allocates until the pop report and the panic.
root = sys.argv[1]


def edit(path, old, new):
    p = root + '/' + path
    s = open(p).read()
    assert s.count(old) == 1, (path, old[:50], s.count(old))
    open(p, 'w').write(s.replace(old, new))


edit('kernel/mm/pmm.c', '''int pmm_free_block_order(uintptr_t addr)
{''', '''/* DEMO ONLY, not committed: clear the order-0 head's next link and leave it. */
void pmm_demo_clear_order0_link(void)
{
    irq_flags_t flags = spin_lock_irqsave(&pmm_lock);
    if (buddy_state.free_lists[0]) {
        buddy_state.free_lists[0]->next = NULL;
    }
    spin_unlock_irqrestore(&pmm_lock, flags);
}
