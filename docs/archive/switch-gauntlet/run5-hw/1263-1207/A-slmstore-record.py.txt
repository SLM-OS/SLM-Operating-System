#!/usr/bin/env python3
"""slmstore-record.py <dest record file> <dir for decoded bytes> <title...>
From this agent's own session transcript, take the most recent labctl
sdwire_ls result for /slmstore (recursive) and the most recent sdwire_cat
(base64) result for each small file, and write: every entry's path, type,
size and mtime, and for the files read, size and sha256 of the decoded bytes.
The decoded bytes are kept in <dir> so that a changed file can be restored."""
import json, sys, base64, hashlib, os, datetime

T = ('/home/john/.claude/projects/-home-john-projects-CS-496-Capstone-SLM-Operating-System'
     '--claude-worktrees-Lab-Session-1263/564b3ef4-5b80-47c5-8d15-6720773a806c/subagents/'
     'agent-abuilder-a-bc5feca53ddd7f51.jsonl')
dst, bdir, title = sys.argv[1], sys.argv[2], ' '.join(sys.argv[3:])
ls = None
cats = {}
for line in open(T, encoding='utf-8', errors='replace'):
    try:
        o = json.loads(line)
    except Exception:
        continue
    content = (o.get('message') or {}).get('content')
    if not isinstance(content, list):
        continue
    for c in content:
        if not isinstance(c, dict) or c.get('type') != 'tool_result':
            continue
        body = c.get('content')
        texts = [body] if isinstance(body, str) else [b.get('text', '') for b in body if isinstance(b, dict)] if isinstance(body, list) else []
        for t in texts:
            try:
                r = json.loads(json.loads(t)['result'])
            except Exception:
                continue
            if not isinstance(r, dict) or r.get('sbc_name') != 'pi-5-1':
                continue
            # The tool result itself carries no time; the transcript entry may. Without one, the
            # record states when it was written: it is always written right after the reads.
            ts = o.get('timestamp') or ('no timestamp in the transcript entry; record written right after, at '
                                        + datetime.datetime.now(datetime.timezone.utc).strftime('%FT%TZ'))
            if r.get('path') == '/slmstore' and 'entries' in r:
                ls = (ts, r)
            elif 'content' in r and r.get('encoding') == 'base64' and str(r.get('path', '')).startswith('/slmstore/'):
                cats[r['path']] = (ts, r)
if not ls:
    sys.exit('no sdwire_ls /slmstore result found')
os.makedirs(bdir, exist_ok=True)
out = ['# ' + title,
       '# source: labctl sdwire_ls (partition 1, /slmstore, recursive) and sdwire_cat (base64) results of this session, from the agent transcript',
       '# listing taken %s; written %s' % (ls[0], datetime.datetime.now(datetime.timezone.utc).strftime('%FT%TZ')),
       '# files.lfs and files.lfs.delta are listed only (not read: an 8 MB sdwire_cat disconnected labctl earlier in the session)',
       'LIST truncated=%s' % ls[1].get('truncated')]
for e in sorted(ls[1]['entries'], key=lambda e: e['path']):
    out.append('ENTRY %s type=%s size=%s mtime=%s' % (e['path'], e['type'], e['size'], e['mtime']))
for p in sorted(cats):
    ts, r = cats[p]
    data = base64.b64decode(r['content'])
    open(os.path.join(bdir, p.strip('/').replace('/', '__')), 'wb').write(data)
    out.append('HASH %s size=%d decoded=%d sha256=%s (read %s)' % (p, r['size'], len(data), hashlib.sha256(data).hexdigest(), ts))
open(dst, 'w').write('\n'.join(out) + '\n')
print('\n'.join(l for l in out if not l.startswith('#')))
