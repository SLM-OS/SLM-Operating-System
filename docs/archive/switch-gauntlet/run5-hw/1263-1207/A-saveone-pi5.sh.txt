#!/bin/bash
# saveone-pi5.sh (Pi 5 evidence worktree; pi-5-1 prints no firmware line on this UART, so the start check accepts the kernel banner block) <labctl output dir name> <evidence file name> <trigger log name in $S/a or -> <title...>
# Saves run_01.txt of one single-boot boot_test call (this session's own output)
# with save.sh (both-ends check), copies the trigger log next to it, and prints
# the R3 counts when the file is an R3 wake boot.
S=/tmp/claude-1000/-home-john-projects-CS-496-Capstone-SLM-Operating-System/564b3ef4-5b80-47c5-8d15-6720773a806c/scratchpad/a
E=/home/john/projects/CS-496-Capstone-SLM-Operating-System/.claude/worktrees/Lab-Session-1263/.wt/a-evp/docs/archive/switch-gauntlet/run5-hw/1263-1207
mkdir -p $E
dir=$1; dst=$2; trig=$3; shift 3
START="${START:-={20,}|SLM-OS v}" bash $S/save.sh /var/lib/labctl/output/$dir/run_01.txt $E/$dst "$*"
if [ "$trig" != - ] && [ -f "$S/$trig" ]; then cp "$S/$trig" $E/; sed 's/^/  trigger: /' "$S/$trig"; fi
case "$dst" in *r3w*) bash $S/r3sum.sh $E/$dst;; esac
