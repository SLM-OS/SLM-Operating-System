#!/bin/bash
# save.sh <labctl run file> <dest evidence file> <title...>
# Run 5's B-save-v2.sh with "poison" lines added to the summary. Copies one labctl
# boot_test run file (this session's own output) verbatim under a header and checks
# both ends: the first 2000 bytes must hold the power-on line (MB1 for Jetson, or
# $START), the SLM-OS banner must be present; the last line is reported.
src=$1; dst=$2; shift 2; title="$*"
START=${START:-'MB1 \(version|Boot-mode|RPi: BOOTSYS|NOTICE:  BL31'}
[ -s "$src" ] || { echo "MISSING $src"; exit 1; }
banner=$(grep -a -m1 -oE 'SLM-OS v[0-9.]+ build [0-9]{14} \([0-9a-f]+(-dirty)?\)' "$src")
first=$(head -c 2000 "$src" | grep -a -m1 -E "$START")
last=$(tail -n 1 "$src" | tr -d '\r' | cut -c1-120)
lines=$(wc -l < "$src")
{
  echo "# $title"
  echo "# source: $src (labctl output of this session, copied verbatim below the header)"
  echo "# banner: ${banner:-NOT FOUND}"
  echo "# power-on line (within first 2000 bytes): ${first:-NOT FOUND}"
  echo "# last line: $last"
  echo "# lines: $lines"
  grep -a -E '^\[FILTER\]|[Pp]oison|\[pmm-verify\] (FINDING|order)|\[pmm-check\] [a-z_: ]*(INCONSISTENT|init:|partial|buddy_free|free_list|pmm_free|walk:|region)|PMM: (order|free)|\[FAIL\] test_|NOT-FINISHED|total failure|All test suites passed|System halted|KERNEL PANIC|PAGE FAULT|PMM: |Boot source' "$src" | tr -d '\r' | cut -c1-220 | sed 's/^/# summary: /' | head -80
  echo "# ---- serial ----"
  cat "$src"
} > "$dst"
ok=OK; [ -z "$banner" ] && ok=NO-BANNER; [ -z "$first" ] && ok="$ok/NO-POWER-ON-LINE"
echo "$(basename "$dst") | $ok | ${banner:-none} | lines=$lines | last=[$last]"
