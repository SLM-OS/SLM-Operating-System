#!/bin/bash
# make test on A (d7ede5d0) alone, then /ci on A+B (37939418), with a
# second WORK_STEALING=ON run if the first fails (#1219).
export PATH=/opt/arm-gnu-toolchain/bin:$HOME/.cargo/bin:$PATH
W=/home/john/projects/CS-496-Capstone-SLM-Operating-System/.claude/worktrees/agent-a8e4eba4c69b43ffd
S=/tmp/claude-1000/-home-john-projects-CS-496-Capstone-SLM-Operating-System--claude-worktrees-bridge-cse-014Q9aryGYwccZqUzMHyre5U/10c489d6-554a-53cc-8bab-69cf599b9e5c/scratchpad
cd $W || exit 1
L=$S/test-A-d7ede5d0; mkdir -p $L
git checkout -q --detach d7ede5d0 && git log --oneline -1 > $L/head.txt
make kernel-test-clean > /dev/null 2>&1
make test > $L/make-test.log 2>&1; echo "make test rc=$?" > $L/status.txt
git checkout -q fixc-work
bash $S/run_ci.sh $S/ci-37939418
L=$S/ci-37939418
if ! grep -q "x-test-ws rc=0" $L/status.txt; then
  make kernel-test-clean > /dev/null 2>&1
  make test WORK_STEALING=ON > $L/x-test-ws-rerun.log 2>&1
  echo "$(date -u +%H:%M:%SZ) x-test-ws-rerun rc=$?" >> $L/status.txt
fi
echo ALLDONE >> $L/status.txt
