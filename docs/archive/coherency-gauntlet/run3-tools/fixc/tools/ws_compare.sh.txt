#!/bin/bash
# Compare the QEMU WORK_STEALING=ON test kernel's sched.c/smp.c objects at A
# (d7ede5d0) and A+B (37939418), then run make test WORK_STEALING=ON twice
# more on A+B.
export PATH=/opt/arm-gnu-toolchain/bin:$HOME/.cargo/bin:$PATH
W=/home/john/projects/CS-496-Capstone-SLM-Operating-System/.claude/worktrees/agent-a8e4eba4c69b43ffd
L=/tmp/claude-1000/-home-john-projects-CS-496-Capstone-SLM-Operating-System--claude-worktrees-bridge-cse-014Q9aryGYwccZqUzMHyre5U/10c489d6-554a-53cc-8bab-69cf599b9e5c/scratchpad/ws-compare
OD=/opt/arm-gnu-toolchain/bin/aarch64-none-elf-objdump
mkdir -p $L
cd $W || exit 1
for c in d7ede5d0 37939418; do
  git checkout -q --detach $c
  make kernel-test-clean > /dev/null 2>&1
  make test WORK_STEALING=ON > $L/test-ws-$c.log 2>&1
  echo "$c make test WORK_STEALING=ON rc=$?" >> $L/status.txt
  for o in sched smp; do
    f=$(find build/kernel-test -name "$o.c.obj" -path "*kernel/sched/*" | head -1)
    $OD -d --no-show-raw-insn $f | sed 1,3d > $L/$o-$c.dis
  done
  sha256sum build/kernel-test/slmos.elf >> $L/status.txt
done
for o in sched smp; do
  echo "$o: $(diff $L/$o-d7ede5d0.dis $L/$o-37939418.dis | grep -c '^[<>]') differing disassembly lines" >> $L/status.txt
done
make kernel-test-clean > /dev/null 2>&1
make test WORK_STEALING=ON > $L/test-ws-37939418-b.log 2>&1
echo "37939418 make test WORK_STEALING=ON (second) rc=$?" >> $L/status.txt
git checkout -q fixc-work
echo DONE >> $L/status.txt
