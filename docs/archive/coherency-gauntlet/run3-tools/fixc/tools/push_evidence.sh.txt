#!/bin/bash
# Add files to an evidence branch without checking it out, then plain push.
# Usage: push_evidence.sh <branch> <repo-dir> <message> <src>...
# Each src lands at <repo-dir>/<basename src>. Fetches the branch tip first
# (the equivalent of pull --rebase for an add-only commit), retries once if
# the push is rejected because someone else pushed in between.
set -e
W=/home/john/projects/CS-496-Capstone-SLM-Operating-System/.claude/worktrees/agent-a8e4eba4c69b43ffd
BR=$1; DIR=$2; MSG=$3; shift 3
cd $W
IDX=$(mktemp -p /tmp/claude-1000/-home-john-projects-CS-496-Capstone-SLM-Operating-System--claude-worktrees-bridge-cse-014Q9aryGYwccZqUzMHyre5U/10c489d6-554a-53cc-8bab-69cf599b9e5c/scratchpad idx.XXXX)
for attempt in 1 2 3; do
  git fetch -q origin "$BR"
  TIP=$(git rev-parse FETCH_HEAD)
  rm -f $IDX
  GIT_INDEX_FILE=$IDX git read-tree $TIP
  for f in "$@"; do
    h=$(git hash-object -w "$f")
    GIT_INDEX_FILE=$IDX git update-index --add --cacheinfo 100644,$h,"$DIR/$(basename "$f")"
  done
  T=$(GIT_INDEX_FILE=$IDX git write-tree)
  C=$(printf '%s\n\nCo-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>\n' "$MSG" | git commit-tree $T -p $TIP)
  if git push -q origin $C:refs/heads/$BR; then
    echo "pushed $C on $BR (parent $TIP)"
    rm -f $IDX
    exit 0
  fi
  sleep 3
done
echo "push failed" >&2
exit 1
