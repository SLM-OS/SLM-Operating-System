#!/bin/bash
# Re-parent a commit range from OLDBASE onto NEWBASE, replacing FILES in each tree with NEWBASE's copies.
set -euo pipefail
OLDBASE=$1 NEWBASE=$2 TIP=$3; shift 3; FILES=("$@")
export GIT_INDEX_FILE=$(mktemp -u)
declare -A MAP; MAP[$(git rev-parse $OLDBASE)]=$(git rev-parse $NEWBASE)
for c in $(git rev-list --reverse --topo-order $OLDBASE..$TIP); do
  git read-tree $c
  for f in "${FILES[@]}"; do
    git update-index --cacheinfo "$(git ls-tree $NEWBASE -- "$f" | awk '{print $1","$3}')","$f"
  done
  tree=$(git write-tree)
  pargs=(); for p in $(git rev-list --parents -n1 $c | cut -d' ' -f2-); do pargs+=(-p "${MAP[$p]:-$p}"); done
  export GIT_AUTHOR_NAME="$(git log -1 --format=%an $c)" GIT_AUTHOR_EMAIL="$(git log -1 --format=%ae $c)" GIT_AUTHOR_DATE="$(git log -1 --format=%aD $c)"
  export GIT_COMMITTER_NAME="$(git log -1 --format=%cn $c)" GIT_COMMITTER_EMAIL="$(git log -1 --format=%ce $c)" GIT_COMMITTER_DATE="$(git log -1 --format=%cD $c)"
  MAP[$c]=$(git log -1 --format=%B $c | git commit-tree $tree "${pargs[@]}")
done
rm -f "$GIT_INDEX_FILE"; echo "${MAP[$(git rev-parse $TIP)]}"
