===== counting script intcount.py
#!/usr/bin/env python3
"""intcount.py <known-list> <file>...

Counts integration-suite runs as C1 counted them (run 3, step6-boots.txt):
a run passes when test_slm_stats_live_from_second_session,
test_component_cleanup_ctx_is_per_task and test_task_lifecycle each print
[PASS], no '[test_wait]' TIMEOUT line appears, and the suite completed
('All test suites passed' or 'total failure(s)'). Failures beyond the known
list (the 18 #1109 tests plus the two detail/wrapper lines) are named.
"""
import re, sys

THREE = ["test_slm_stats_live_from_second_session",
         "test_component_cleanup_ctx_is_per_task",
         "test_task_lifecycle"]
known = set(l.strip() for l in open(sys.argv[1]) if l.strip())
print("| file | banner | completed | total failures | 3 tests PASS | test_wait TIMEOUT | panic | beyond known | known missing | verdict |")
print("|---|---|---|---|---|---|---|---|---|---|")
npass = 0
files = sys.argv[2:]
for f in files:
    t = open(f, encoding="utf-8", errors="replace").read()
    ban = re.findall(r"SLM-OS v\S+ build \S+ \((\w+)\)", t)
    tot = re.findall(r"\[FAIL\] (\d+) total failure", t)
    allp = "All test suites passed" in t
    completed = bool(tot) or allp
    fails = set()
    for m in re.finditer(r"^\s*\[FAIL\] (.+?)\s*$", t, re.M):
        name = m.group(1)
        if re.match(r"\d+ (test\(s\) failed|total failure)", name):
            continue
        fails.add(name)
    three = {n: bool(re.search(r"\[PASS\] " + n + r"\s*$", t, re.M)) for n in THREE}
    tw = len(re.findall(r"\[test_wait\].*TIMEOUT", t))
    pan = bool(re.search(r"KERNEL PANIC|PAGE FAULT|System halted", t))
    beyond = sorted(fails - known)
    missing = sorted(known - fails)
    ok = completed and all(three.values()) and tw == 0 and not pan
    npass += ok
    print(f"| {f.split('/')[-1]} | {','.join(sorted(set(ban))) or 'NONE'} | {completed} | "
          f"{tot[-1] if tot else ('0' if allp else '-')} | "
          f"{sum(three.values())}/3{'' if all(three.values()) else ' missing ' + ','.join(n for n, v in three.items() if not v)} | "
          f"{tw} | {pan} | {'; '.join(beyond) or 'none'} | {len(missing)} | {'pass' if ok else 'FAIL'} |")
print(f"# {npass}/{len(files)} pass by C1's rule")
