# #1193 coherency gauntlet: Raspberry Pi OS kernel + litmus initramfs.
# Staged on the SLM-OS card as config.txt only for the duration of the run.
# No armstub line: the stock BL31 embedded in the EEPROM firmware runs,
# the same one SLM-OS runs under.
arm_64bit=1
kernel=coh1193_kernel.img
initramfs coh1193_initrd.gz followkernel
device_tree=coh1193_rpi5.dtb
cmdline=coh1193_cmdline.txt
enable_uart=1
uart_2ndstage=1
# The packaged DTB ships RP1 uart0 (GPIO 14/15, ttyAMA0) disabled; this
# base-DT override enables it and points the console alias at it.
dtparam=uart0_console
# pciex4_reset is left at the Raspberry Pi OS default: results come over
# HTTP (RP1 ethernet), so no earlycon needs RP1's BAR before Linux
# re-trains the link.
